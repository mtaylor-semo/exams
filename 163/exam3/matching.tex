\question\MatchQuestion{Corridor}{Can reduce local extinction by connecting sources and sinks.}
\question\MatchQuestion{Commensalism}{One species gets ecological benefit over the long term, the other neither benefits or loses.}
\question\MatchQuestion{Richness}{The total number of species present in the community.} 
\question\MatchQuestion{Paedomorphy}{Retention of child-like traits in adult organism.}
\question\MatchQuestion{Aposomatic Coloration}{A poisonous species warns a potential predator.}
\question\MatchQuestion{Pseudogene}{Duplicated gene loses its function.}
\question\MatchQuestion{Keystone Species}{A species that greatly increases community diversity.}
\question\MatchQuestion{Realized niche}{Based on the resources actually used by a species.}
\question\MatchQuestion{Early succession}{Composed mostly of grasses and other generalist species.}
\question\MatchQuestion{Emigration}{Individuals leave a population.}
