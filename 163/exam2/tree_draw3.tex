\question[15]
The table shows the presence and absence of six homologies (1–6) among seven species (A–G). Use these data to draw a phylogenetic tree. Draw a tick mark across the branch of each common ancestor and label the homology that unites that clade.

\vspace*{\baselineskip}

\begin{longtable}{@{}lrrrrrr@{}}
\toprule
	&	1	&	2	&	3	&	4	&	5	& 6 \tabularnewline
\midrule
A	&	1	&	0	&	0	&	1	&	1	&	1 \tabularnewline
B	&	0	&	0	&	0	&	0	&	0	&	1 \tabularnewline
C	&	1	&	1	&	0	&	0	&	0	&	1 \tabularnewline
D	&	1	&	0	&	1	&	1	&	0	&	1 \tabularnewline
E	&	1	&	1	&	0	&	0	&	0	&	1 \tabularnewline
F	&	1	&	0	&	1	&	1	&	0	&	1 \tabularnewline
G	&	1	&	0	&	0	&	1	&	1	&	1 \tabularnewline
\bottomrule
\end{longtable}

\vspace*{2\baselineskip}
\begin{AnswerPage}{5in}

\centering

\begin{forest}
  mytree,
    [[,name=ALL
		[,name=AGDFC
			[,name=AGDF
				[,name=AG
					[A]
					[G]
				]
				[,name=DF
					[D]
					[F]
				]
			]
			[,name=CE
				[C]
				[E]
			]
		]
		[B]
    ]]
    \draw [thick] ([xshift=-2mm, yshift=-5mm] AG.base)--+(0:4mm) node [right] {5};
    \draw [thick] ([xshift=-2mm, yshift=-5mm] DF.base)--+(0:4mm) node [right] {3};
    \draw [thick] ([xshift=-2mm, yshift=-5mm] AGDF.base)--+(0:4mm) node [right] {4};
    \draw [thick] ([xshift=-2mm, yshift=-5mm] CE.base)--+(0:4mm) node [right] {2};
    \draw [thick] ([xshift=-2mm, yshift=-5mm] AGDFC.base)--+(0:4mm) node [right] {1};
    \draw [thick] ([xshift=-2mm, yshift=-5mm] ALL.base)--+(0:4mm) node [right] {6};
\end{forest}

\end{AnswerPage}
